// Task 5 (pqzl-fs3, 2026-10-19): re-prove pqz-core/fuzz/regress/valid/crash-b9881dfc… (the pre-vector harness's
// DISCLOSE-RECEIVER fixture input, PQZL v2) under PQZL v3: same statement (fixture::cases, id now v3), same rnd32 = 0x33..
// Usage: refix <old input> <new input out>
use pqz_ffi::*;
fn main() {
    let a: Vec<String> = std::env::args().collect();
    let old = std::fs::read(&a[1]).unwrap();
    println!("old: decode {:?}", stmt::decode_input(&old).err());
    let n = u32::from_be_bytes(old[1..5].try_into().unwrap()) as usize;
    let (op, s, pr) = (old[0], &old[5..5 + n], &old[5 + n..]);
    println!("old: op {op} stmt {n} B proof {} B, id {}", pr.len(), s[32..64].iter().map(|b| format!("{b:02x}")).collect::<String>());
    println!("old: precompile word {:02x?} (v2 id: unknown proofSystemId under v3)", precompile(&old).map(|w| w[31]));
    let c = fixture::cases(&PQZL_ID).unwrap().into_iter().find(|c| c.form == stmt::Form::DiscloseReceiver).unwrap();
    let mut s_old = s.to_vec();
    s_old[32..64].copy_from_slice(&PQZL_ID);
    println!("statement = fixture::cases(PQZL v3) DiscloseReceiver with the id word replaced: {}", s_old == c.stmt);
    assert_eq!(s_old, c.stmt);
    let p = prove(c.op(), &c.stmt, &c.wit, &[0x33; 32]).unwrap();
    let mut x = vec![c.op()];
    x.extend_from_slice(&(c.stmt.len() as u32).to_be_bytes());
    x.extend_from_slice(&c.stmt);
    x.extend_from_slice(&p);
    let w = precompile(&x).unwrap();
    println!("new: proof {} B, input {} B, precompile word {:02x}, generic verifier {:?}", p.len(), x.len(), w[31], pqzl::verify_generic(c.op(), &c.stmt, &p));
    assert_eq!(w[31], 1);
    let p2 = prove(c.op(), &c.stmt, &c.wit, &[0x33; 32]).unwrap();
    println!("re-proved again: byte-equal {}", p2 == p);
    std::fs::write(&a[2], &x).unwrap();
}
