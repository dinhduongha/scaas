(cd $L && docker compose stop blockchain-node)            # only if running

# C1. Backup of the stack (chainspec, config, compose, password, keystore, chain data; data is root-owned: alpine)
docker run --rm --memory 256m -v $L:/l alpine:3 sh -c "set -e; B=/l/backup-pqz5-$TS; mkdir \$B
  cp -a /l/genesis.json /l/config.json /l/docker-compose.yml /l/password.txt /l/keystore /l/data/nethermind_data \$B/
  (cd /l/data/nethermind_data && find . -type f | sort | xargs sha256sum) > /tmp/a
  (cd \$B/nethermind_data && find . -type f | sort | xargs sha256sum) > /tmp/b
  cmp /tmp/a /tmp/b && echo \"data backup identical: \$(wc -l < /tmp/a) files\"; du -s /l/data/nethermind_data \$B"

# C2. Chainspec: ONLY from $L/genesis.json (7702 off; never a scratchpad genesis copy)
grep -c 7702 $L/genesis.json                              # must print 0

  $S/genesis.pqz6.json --testnet-reset                    # expect: engine.Pqz replaced, params changed: none, genesis header changed: none
grep -c 7702 $S/genesis.pqz6.json                         # must print 0

# C3. Install: genesis, compose image pqz5 -> pqz6, empty chain data (backup in C1)
